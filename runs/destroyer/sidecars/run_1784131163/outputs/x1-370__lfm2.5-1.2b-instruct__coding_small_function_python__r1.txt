```python
import re

def normalize_phone_number(value: str) -> str:
    """
    Normalizes a phone number according to the following rules:
    1. Only digits are kept.
    2. Supports 10-digit US numbers (e.g., "1234567890").
    3. Supports 11-digit numbers beginning with '1' (e.g., "1000000001", "011234567890").
    
    Parameters
    ----------
    value : str
        The input phone number string.
        
    Returns
    -------
    str
        A normalized version of the phone number.
        
    Raises
    ------
    ValueError
        If the input does not match any valid format.
    """
    # Remove non-digit characters
    cleaned = re.sub(r'\D', '', value)
    
    # Check length requirements
    if len(cleaned) == 10 and cleaned.startswith('1'):
        return cleaned
    elif len(cleaned) == 11:
        return cleaned[0] + cleaned[1:]
    else:
        raise ValueError("Phone number must be either 10 digits or start with '1' followed by 10 digits.")

if __name__ == '__main__':
    # Test cases
    assert normalize_phone_number("1234567890") == "1234567890"
    assert normalize_phone_number("12345678901") == "12345678901"  # 11-digit with leading 1
    assert normalize_phone_number("1000000001") == "1000000001"
    assert normalize_phone_number("011234567890") == "1234567890"  # starts with 1 and 10 digits
    assert normalize_phone_number("abc123def") == ValueError
    assert normalize_phone_number("555-0123") == "5550123"
    
    print("All tests passed!")
```
