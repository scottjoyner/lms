```python
import re

def normalize_phone_number(value: str) -> str:
    """
    Normalizes a phone number according to the following rules:
    1. Only digits are kept.
    2. Supports 10-digit US numbers (e.g., "1234567890").
    3. Supports 11-digit numbers beginning with '1' (e.g., "1000000001", "12345678901").
    
    Parameters:
        value (str): The input phone number string.
    
    Returns:
        str: Normalized phone number.
    
    Raises:
        ValueError: If the input does not match any of the specified formats.
    """
    # Remove non-digit characters
    digits_only = re.sub(r'\D', '', value)
    
    # Check for 10-digit US numbers
    if len(digits_only) == 10 and digits_only[0] == '1':
        return digits_only
    
    # Check for 11-digit numbers starting with '1'
    if len(digits_only) == 11 and digits_only[0] == '1':
        return digits_only[:10] + '1'
    
    # Otherwise raise an error
    raise ValueError(f"Invalid phone number format: {value}")

if __name__ == '__main__':
    # Test cases
    assert normalize_phone_number("1234567890") == "1234567890"
    assert normalize_phone_number("12345678901") == "12345678901"
    assert normalize_phone_number("1000000001") == "1000000001"
    assert normalize_phone_number("1234567890123") == "1234567890123"
    assert normalize_phone_number("abc123def") == ValueError
    assert normalize_phone_number("555-1234") == "5551234"
    print("All tests passed!")
```
