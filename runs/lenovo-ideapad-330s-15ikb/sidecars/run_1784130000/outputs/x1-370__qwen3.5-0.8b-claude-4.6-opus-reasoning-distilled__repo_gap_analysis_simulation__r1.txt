## Docker Configuration Plan

```dockerfile
# docker-compose.yml
version: '3.8'
services:
  backend:
    build: .
    ports:
      - "5000:5000"
    environment:
      - DATABASE_URL=sqlite:///backend.db
      - JWT_SECRET=${JWT_SECRET}
      - API_KEY=${API_KEY}
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5000/health"]
      interval: 30s
      timeout: 10s
      retries: 3
    restart: unless-stopped

  frontend:
    build: .
    ports:
      - "8080:8080"
    depends_on:
      - backend
```
