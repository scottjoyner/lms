import re

def normalize_phone_number(value: str) -> str:
    """
    Normalize a phone number according to the following rules:
    1. Remove all non-digit characters.
    2. Support 10-digit US numbers (e.g., "1234567890") and 11-digit numbers starting with '1'
       (e.g., "10000000001", "1234567890123").
    3. Return the normalized string.
    
    Raises ValueError if the input does not match any of the expected patterns.
    """
    # Remove all non-digit characters
    cleaned = re.sub(r'\D', '', value)
    
    # Check for 10-digit US numbers (10 digits total, first digit optional)
    if len(cleaned) == 10 and cleaned[0] == '0':
        return cleaned
    elif len(cleaned) == 11 and cleaned.startswith('1'):
        return cleaned
    
    # If no valid pattern matches, raise an error
    raise ValueError(f"Invalid phone number format: {value}")

if __name__ == '__main__':
    # Test cases with assertions
    assert normalize_phone_number("1234567890") == "1234567890"
    assert normalize_phone_number("1234567890123") == "1234567890123"
    assert normalize_phone_number("12345678901") == "12345678901"
    assert normalize_phone_number("123456789") == "123456789"  # 10-digit
    assert normalize_phone_number("0000000000") == "0"      # edge case
    assert normalize_phone_number("abc123xyz") == "123"
    assert normalize_phone_number("555-123-4567") == "555-123-4567"
    assert normalize_phone_number("1-800-555-0199") == "800-555-0199"
    
    print("All tests passed!")
