### 2. Authentication (Critical)

# GOOD - JWT token with expiration
from jose import jwt, verify_signature
import os

TOKEN_FILE = "/tmp/admin_token.json"

@app.post("/token", response_model=jwt.Token)
async def create_token(request):
    data = request.headers.get("Authorization")
    
    if not data:
        return {"error": "Unauthorized"}, 401
    
    token = jwt.decode(data, os.environ["SECRET_KEY"], algorithms=["HS256"])
    
    # Token expiration (e.g., 30 minutes)
    expires_at = int(os.environ.get("EXPIRE_AT", 1800))
    
    return {
        "token": token,
        "expires_in": expires_at,
        "scope": request.headers.get("X-Auth-Scope") or ""
    }
