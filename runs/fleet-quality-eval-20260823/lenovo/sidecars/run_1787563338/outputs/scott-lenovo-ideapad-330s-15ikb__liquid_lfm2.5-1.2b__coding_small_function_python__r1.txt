Here's a Python function that matches your requirements:

- Strips non-digits from the input string.
- Accepts 10-digit US numbers (e.g., "1234567890") and 11-digit numbers starting with "1" (e.g., "1000000001").
- Raises ValueError for anything else.
- Includes a test block with assertions.

```python
import re

def normalize_phone_number(value: str) -> str:
    """
    Normalizes a phone number string according to the following rules:
    1. Removes all non-digit characters.
    2. Accepts 10-digit US numbers (e.g., "1234567890").
    3. Accepts 11-digit numbers starting with '1' (e.g., "1000000001").
    4. Raises ValueError for any other format.

    Parameters:
        value (str): Input phone number string.

    Returns:
        str: Normalized phone number.

    Raises:
        ValueError: If the input does not match the expected patterns.
    """
    # Remove non-digit characters
    cleaned = re.sub(r'\D', '', value)
    
    # Check for 10-digit US numbers (10 digits total, first digit 1-9)
    if len(cleaned) == 10 and cleaned[0] != '1':
        raise ValueError("Number must start with '1' for 11-digit format.")
    if len(cleaned) != 10:
        raise ValueError("Number must be exactly 10 digits.")

    # Check for 11-digit numbers starting with '1'
    if len(cleaned) == 11 and cleaned[0] == '1':
        return cleaned
    else:
        raise ValueError("Invalid phone number format.")

# Test block
if __name__ == '__main__':
    tests = [
        ("1234567890", "1234567890"),         # 10-digit US
        ("12345678901", "12345678901"),      # 11-digit starting with 1
        ("1000000001", "1000000001"),        # 11-digit with leading zeros allowed
        ("abc123def", ValueError),             # Invalid non-digits
        ("1234", ValueError),                   # Too short
        ("1234567890123", ValueError)          # Too long
    ]

    for input_str, expected in tests:
        result = normalize_phone_number(input_str)
        assert result == expected, f"Failed for {input_str}: expected {expected}, got {result}"
    
    print("All tests passed!")
```
