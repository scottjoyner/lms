def normalize_phone_number(value: str) -> str:
    # Strip non-digits
    digits = ''.join(ch for ch in value if ch.isdigit())
    # If empty or not 10 nor 11 digits => error
    if len(digits) == 0:
        raise ValueError("No digits found")
    if len(digits) == 10:
        return digits
    elif len(digits) == 11 and digits[0] == '1':
        # Remove country code
        return digits[1:]
    else:
        raise ValueError(f"Invalid phone number: {value!r}")

- Input with spaces, dashes, parentheses, etc. Should strip them.
- If input is already just digits but length not 10 or 11 => error.
