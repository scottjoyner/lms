import re

def normalize_phone_number(value: str) -> str:
    # Remove all non-digit characters from the input string
    cleaned_value = re.sub(r'\D', '', value)
    
    # Check if the cleaned number is a 10-digit US number or an 11-digit number starting with 1
    if len(cleaned_value) == 10 and cleaned_value[0] == '1':
        return cleaned_value
    elif len(cleaned_value) == 11:
        return cleaned_value[:-1]
    else:
        raise ValueError("Invalid phone number format. Expected a 10-digit US number or an 11-digit number starting with 1.")

if __name__ == '__main__':
    # Test cases
    assert normalize_phone_number('(123) 456-7890') == '1234567890'
    assert normalize_phone_number('1-234-567-890') == '1234567890'
    assert normalize_phone_number('12345678901') == '12345678901'
    assert normalize_phone_number('1234567890123') == '1234567890123'
    try:
        normalize_phone_number('invalid')
    except ValueError as e:
        assert str(e) == "Invalid phone number format. Expected a 10-digit US number or an 11-digit number starting with 1."
