def normalize_phone_number(value: str) -> str:
    digits = ''.join(char for char in value if char.isdigit())

    if len(digits) == 10:
        return digits
    elif len(digits) == 11 and digits[0] == '1':
        return digits
    else:
        raise ValueError(f"Invalid phone number: {value}")


if __name__ == '__main__':
    assert normalize_phone_number("555-123-4567") == "5551234567"
    assert normalize_phone_number("+1-555-123-4567") == "15551234567"
    assert normalize_phone_number("(800) 555-1234") == "8005551234"
    assert normalize_phone_number("1 (800) 555-1234") == "18005551234"

    try:
        normalize_phone_number("123")
        assert False, "Should have raised ValueError for invalid input"
    except ValueError:
        pass

    print("All tests passed!")
