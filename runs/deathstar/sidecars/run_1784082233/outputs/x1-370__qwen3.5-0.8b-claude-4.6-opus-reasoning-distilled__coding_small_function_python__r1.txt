import re
from typing import Tuple, Optional


def normalize_phone_number(value: str) -> str:
    """
    Strips non-digits from a phone number string.
    
    Supports 10-digit US numbers (e.g., 555-555-5555) and 11-digit numbers 
    beginning with 1 (e.g., 111-123-4567). Raises ValueError for anything else.
    
    Args:
        value: Phone number string to process
        
    Returns:
        Clean phone number string
        
    Raises:
        ValueError: If the input is not a valid phone number
    """
    # Pattern to match 10-digit US numbers (e.g., 555-555-5555)
    # This pattern ensures we capture digits and separators
    ten_digit_pattern = re.compile(r'\d{3}-\d{4}')
    
    # Pattern to match 11-digit numbers beginning with 1 (e.g., 111-123-4567)
    # This pattern ensures we capture digits and separators
    eleven_digit_pattern = re.compile(r'\d{3}-\d{4}')
    
    # Combine patterns to match valid phone numbers
    combined_pattern = ten_digit_pattern + eleven_digit_pattern
    
    # Apply the combined pattern to validate phone numbers
    result = combined_pattern.sub('', value)
    
    return result
