def normalize_phone_number(value: str) -> str:
    # Strip non-digits
    digits = ''.join(ch for ch in value if ch.isdigit())
    
    # Accept 10-digit numbers (US)
    if len(digits) == 10:
        return digits
    
    # Accept 11-digit numbers beginning with '1' (international format: +1 (xxx) xxx-xxxx or similar)
    if len(digits) == 11 and digits[0] == '1':
        return digits[1:]  # strip the leading 1
        
    raise ValueError(f"Invalid phone number: {value!r}")
    
That's it.
