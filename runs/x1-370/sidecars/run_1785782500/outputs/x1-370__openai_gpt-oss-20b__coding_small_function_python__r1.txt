import re

def normalize_phone_number(value: str) -> str:
    """
    Normalize a phone number string by removing all non-digit characters.
    
    Supports:
        - 10-digit US numbers (e.g., "202-555-0123" -> "2025550123")
        - 11-digit numbers beginning with '1' (e.g., "+1 (202) 555-0123" -> "12025550123")
    
    Raises:
        ValueError: If the resulting number does not match one of the supported formats.
    """
    digits = re.sub(r'\D', '', value)
    if len(digits) == 10:
        return digits
    if len(digits) == 11 and digits.startswith('1'):
        return digits
    raise ValueError(f"Invalid phone number format: {value!r}")

if __name__ == "__main__":
    # Valid 10-digit numbers
    assert normalize_phone_number("202-555-0123") == "2025550123"
