**1. User Model Design**
```python
class User(Base):
    __tablename__ = 'users'
    
    id: int = Field(primary_key=True, index=True)
    username: str = Field(..., min_length=3, unique=True)
    email: EmailStr = Field(..., description="Unique identifier for user account")
    password_hash: str = Field(..., description="Password encrypted with bcrypt")
    is_active: bool = Field(default=True, description="Whether the account is active")
    role: str = Field(..., default='regular', enum=['admin', 'regular'])
    created_at: datetime = Field(..., index=True)
    
    # Additional fields for email verification
    email_verified: Optional[bool] = None
    
    async def save(self, user_data: dict):
        hashed_password = await self.hash_password(user_data['password'])
        self.password_hash = hashed_password
        self.email_verified = False
        self.save()  # Trigger index creation
```

```python
async def generate_token(user_id: int, role: str):
    """Generate secure JWT token"""
    payload = {
        'sub': user_id,
        'iat': datetime.utcnow(),
        'exp': datetime.utcnow() + timedelta(days=1),  # Token expiration
        'roles': role
    }
    
    try:
        token = jwt.encode(payload, SECRET_KEY, algorithm='HS256')
        return token
    except JWTError as e:
        raise HTTPException(status_code=401, detail=str(e))

async def verify_token(token: str) -> Optional[dict]:
    """Verify and decode JWT token"""
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=['HS256'])
        return {
            'sub': payload['sub'],
            'roles': payload.get('roles', [])
        }
    except JWTError as e:
        raise HTTPException(status_code=401, detail=str(e))

async def get_current_user(token: Optional[str]) -> Optional[dict]:
    """Get current authenticated user"""
    if not token:
        return None
    
    payload = jwt.decode(token, SECRET_KEY, algorithms=['HS256'])
    return {
        'id': payload['sub'],
        'username': payload.get('sub', ''),
        'roles': payload.get('roles', [])
    }

async def create_user(user_data: dict) -> User:
    """Create new user"""
    try:
        hashed_password = await Base.metadata.create_table(
            'passwords'
        )
        
        # Verify existing password if email is provided
        if user_data.get('email'):
            existing_user = await Base.query.filter_by(email=user_data['email']).first()
            if existing_user and not existing_user.is_active:
                raise HTTPException(
                    status_code=401, 
                    detail="Account already exists"
                )
        
        user = User(
            username=user_data['username'],
            email=user_data.get('email', ''),
            password_hash=hashed_password,
            is_active=True,
            role='regular'
        )
        await user.save()
        return user
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
