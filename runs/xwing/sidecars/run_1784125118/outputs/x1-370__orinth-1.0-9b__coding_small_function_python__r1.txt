def normalize_phone_number(value: str) -> str:
    digits = ''.join(c for c in value if c.isdigit())

    if len(digits) == 10:
        return digits
    if len(digits) == 11 and digits[0] == '1':
        return digits
    raise ValueError(f"Invalid phone number: {value!r}")


if __name__ == '__main__':
    assert normalize_phone_number("555-123-4567") == "5551234567"
    assert normalize_phone_number("+1-555-123-4567") == "15551234567"
    assert normalize_phone_number("(555) 123-4567") == "5551234567"
    assert normalize_phone_number("1 (800) 555-1234") == "18005551234"
    try:
        normalize_phone_number("123")
        assert False, "Should have raised ValueError for short number"
    except ValueError:
        pass

    try:
        normalize_phone_number("12345678901")
        assert False, "Should have raised ValueError for 11-digit without leading 1"
    except ValueError:
        pass
