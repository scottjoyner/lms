def normalize_phone_number(value: str) -> str:
    # Strip non-digits
    digits = ''.join(ch for ch in value if ch.isdigit())
    # If empty or contains non-digit after stripping? Actually we already have only digits.
    # Validate length and leading 1 optional
    if len(digits) == 0:
        raise ValueError("Empty phone number")
    if len(digits) == 10:
        return digits
    elif len(digits) == 11 and digits[0] == '1':
        return digits[1:]  # strip country code
    else:
        raise ValueError(f"Invalid phone number: {value!r}")
